二、State 定义（图的状态契约）

from typing import TypedDict

class AgentState(TypedDict, total=False):
    # 输入
    request: AgentRequest            # message / language / session_id / source
    # assemble_context 产出
    history: list[ConversationMessage]   # 短期会话上下文（有界 LRU）
    identity: IdentitySnapshot           # unknown / guest / owner_verified
    memories: list[MemoryRecord]         # 已确认记忆（仅 owner 非空）
    persona: PersonaProfile | None       # 人设配置
    references: list[RagSearchHit]       # RAG 检索片段（不可信标记）
    rag_status: Literal["disabled", "used", "empty", "unavailable"]
    # 规划与执行产出
    tool_plan: VoiceToolPlan | None      # 语音明确指令的本地规划
    tool_result: ToolCallResult | None   # 受控工具执行结果（含审计）
    # 模型产出
    response: AgentResponse | None       # 结构化响应（贯穿后处理）
    error: AgentProviderError | None     # 模型失败信息（驱动降级条件边）

三、LangGraph 代码（Phase 1 · 接口不变替换内部实现）

from langgraph.graph import StateGraph, START, END

def build_agent_graph(service: AgentService):
    g = StateGraph(AgentState)

    g.add_node("assemble_context",       service.assemble_context)       # 原 respond() 前半段
    g.add_node("plan_voice_tool",        service.plan_voice_tool)        # 原 voice_tool_planner 调用
    g.add_node("execute_controlled_tool", service.run_controlled_tool)   # 原 tool_service.execute
    g.add_node("format_tool_response",   service.format_tool_response)  # 原 _voice_tool_response
    g.add_node("call_remote_agent",       service.call_remote_agent)     # 原 OpenAiCompatibleAgent.respond
    g.add_node("mock_agent",              service.mock_respond)          # 原 MockAgent.respond
    g.add_node("mock_fallback",           service.mock_respond)          # 复用同一实现
    g.add_node("raise_provider_error",    service.raise_provider_error)  # 原 _fallback_or_raise 的抛错分支
    g.add_node("postprocess",             service.postprocess)           # 原 respond() 后半段

    g.add_edge(START, "assemble_context")
    g.add_edge("assemble_context", "plan_voice_tool")

    # 条件边 ①：语音明确指令 / mock / 远程模型
    g.add_conditional_edges("plan_voice_tool", route_input, {
        "tool":  "execute_controlled_tool",
        "mock":  "mock_agent",
        "agent": "call_remote_agent",
    })

    # 条件边 ②：输出校验 → 通过 / 降级 / 抛错
    g.add_conditional_edges("call_remote_agent", validate_output, {
        "postprocess":    "postprocess",
        "mock_fallback":  "mock_fallback",
        "raise":          "raise_provider_error",
    })

    g.add_edge("execute_controlled_tool", "format_tool_response")
    g.add_edge("format_tool_response", "postprocess")
    g.add_edge("mock_agent", "postprocess")
    g.add_edge("mock_fallback", "postprocess")
    g.add_edge("raise_provider_error", END)
    g.add_edge("postprocess", END)

    return g.compile()

# ---- 条件边路由函数 ----
def route_input(state: AgentState) -> str:
    if state.get("tool_plan") is not None:
        return "tool"                      # 明确指令直达受控工具（1 秒级，零 token）
    if settings.agent_provider == "mock":
        return "mock"
    return "agent"

def validate_output(state: AgentState) -> str:
    if state.get("error") is None:
        return "postprocess"
    if settings.agent_fallback_to_mock and not settings.agent_require_remote:
        return "mock_fallback"
    return "raise"

# ---- AgentService 对外接口保持不变 ----
class AgentService:
    def __init__(self, ...):
        ...
        self._graph = build_agent_graph(self)

    async def respond(self, request: AgentRequest) -> AgentResponse:
        state = {"request": request}
        result = await self._graph.ainvoke(state)
        return result["response"]          # 抛错分支在节点内 raise，行为与现状一致

五、Phase 2：引入 LangGraph 原生能力（真正的增量价值）

1. 工具审批的人机协作（interrupt） —— 中风险工具（如 open_file）不再是"生成审批单后异步等待"，而是工作流挂起：

from langgraph.types import interrupt

def execute_controlled_tool(state: AgentState):
    tool = state["tool_plan"].request
    if needs_manual_approval(tool):        # 中风险工具
        decision = interrupt({              # 图在此挂起，等待人工输入
            "tool": tool.tool_name,
            "arguments": tool.arguments,
            "timeout": "10min",
        })
        if decision != "approved":
            return rejected_result(tool)    # 拒绝 → 走不悦动作分支
    return tool_service.execute(tool, state["identity"])

2. checkpointer 检查点 —— 会话状态可持久化恢复（替换进程内 LRU 的同时保留短期记忆语义），网关重启不丢对话现场；

3. astream_events 节点级事件流 —— 前端"阶段状态"面板从手写状态机升级为直接消费图的节点事件，天然对齐现有"等待→聆听→识别→提交→TTS→完成"的展示。
